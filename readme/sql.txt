UPDATING FOLDER CHANGES IN THE DATABASE(IMAGE\ TO IMAGES\)
-- 1. preview: which rows would change?
SELECT post_id, media FROM posts WHERE media LIKE 'image/%';

-- 2. update just one row first, and test it in the browser
UPDATE posts
SET media = REPLACE(media, 'image/', 'images/')
WHERE post_id = 21;

-- 3. if that row's image now shows, update all the others
UPDATE posts
SET media = REPLACE(media, 'image/', 'images/')
WHERE media LIKE 'image/%';
